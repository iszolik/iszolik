│ function buildAjaxUrl(option, task, additionalParams = {}) {             │
│     // 1. Abszolút URL a Joomla gyökérből                                │
│     const baseUrl = window.location.origin + '/index.php';               │
│                                                                           │
│     // 2. Paraméterek                                                    │
│     const params = new URLSearchParams();                                │
│     params.append('option', option);                                     │
│     params.append('task', task);                                         │
│                                                                           │
│     // 3. Itemid megőrzése                                               │
│     const currentUrl = new URL(window.location.href);                    │
│     const itemId = currentUrl.searchParams.get('Itemid');                │
│     if (itemId) params.append('Itemid', itemId);                         │
│                                                                           │
│     // 4. Solidres paraméterek                                           │
│     ['property_id', 'hub_id', 'site_id', 'reservation_id']               │
│         .forEach(param => {                                              │
│             const value = currentUrl.searchParams.get(param);            │
│             if (value) params.append(param, value);                      │
│         });                                                              │
│                                                                           │
│     // 5. További paraméterek + JSON formátum                            │
│     Object.keys(additionalParams).forEach(key => {                       │
│         params.append(key, additionalParams[key]);                       │
│     });                                                                  │
│     params.append('format', 'json');                                     │
│                                                                           │
│     return baseUrl + '?' + params.toString();                            │
│ }                                                                         │
│                                                                           │
